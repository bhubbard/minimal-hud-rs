src/main.rs RUST 
use minimal_hud_rs::{HudState, VitalStats, VehicleTelemetry, DirtyFlags};

fn main() {
    let mut hud = HudState::new();

    // Update vitals on game tick
    hud.update_health(75.0);
    hud.update_hunger(40.0);

    // Zero-cost dirty check: only dispatch when flags are non-empty
    if let Some(dirty_payload) = hud.take_dirty() {
        println!("Dispatching HUD update: {:?}", dirty_payload);
    }
}
